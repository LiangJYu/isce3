 38    def parseFileDescriptor(self, fid):
 39        '''
 40        Parse SAR Leaderfile descriptor record.
 41        '''
 42        from isce3.parsers.CEOS.LeaderFileDescriptorType import LeaderFileDescriptorType
 43
 44        #Description of record - seems to be common across missions
 45        record = LeaderFileDescriptorType()
 46            
 47        #Read the record
 48        record.fromfile(fid)
 49
 50        #Sensor specific validators
 51        #http://www.ga.gov.au/__data/assets/pdf_file/0019/11719/GA10287.pdf Pg: 3-23
 52        assert(record.RecordSequenceNumber == 1)
 53        assert(record.FirstRecordType == 11)
 54        assert(record.RecordTypeCode == 192)
 55        assert(record.SecondRecordSubType == 18)
 56        assert(record.ThirdRecordSubType == 18)
 57        assert(record.RecordLength == 720)
 58        assert(record.RecordSequenceType == 'FSEQ')
 59        assert(record.RecordCodeType == 'FTYP')
 60        assert(record.RecordFieldType == 'FLGT')
 61
 62        #Check length of record
 63        assert(fid.tell() == record.RecordLength)
 64
 65        #Return the validated record
 66        return record

 68    def parseDatasetSummary(self, fid):
 69        '''
 70        Parse Dataset Summary record.
 71        '''
 72        from .DataSetSummaryRecordType import DatasetSummaryRecordType
 73
 74        #First ensure that spec says there is only one record
 75        assert(self.description.NumberDatasetSummaryRecords == 1)
 76        
 77        #Description of record - customized record from local directory
 78        record = DatasetSummaryRecordType()
 79
 80        #Read the record
 81        record.fromfile(fid)
 82
 83        #Sensor specific validators
 84        #http://www.ga.gov.au/__data/assets/pdf_file/0019/11719/GA10287.pdf Pg: 3-26/38
 85        assert(record.RecordSequenceNumber == 2)
 86        assert(record.FirstRecordType == 18)
 87        assert(record.RecordTypeCode == 10)
 88        assert(record.SecondRecordSubType == 18)
 89        assert(record.ThirdRecordSubType == 20)
 90        assert(record.RecordLength == 4096)
 91        assert(record.NumberOfSARChannels in [1,2,4])
 92        assert(record.SensorPlatformMissionIdentifier == 'ALOS')
 93        #Specific check for ALOS stripmap, normal observation mode
 94        assert(record.SensorIDAndMode[0:9] == 'ALOS  -L ')
 95        assert(record.SensorIDAndMode[10] in ["H", "L"])
 96        assert(record.SensorIDAndMode[12:14] in ("60", "62")) # high res or polsar
 97        assert(record.SensorClockAngle == 90.0)
 98        assert(record.RangePulseCodeSpecifier == "LINEAR FM CHIRP")
 99        assert(record.QuantizationDescriptor == "UNIFORM I,Q")
100        assert(record.ProductTypeSpecifier == "UNPROCESSED SIGNAL DATA")
101        assert(record.DataInputSource == "ONLINE")
102        assert(record.LineContentIndicator == "RANGE")
103        assert(0 <= record.AntennaBeamNumber <= 22)
104        assert(0 <= record.ParameterTableNumber <= 191)
105        assert(record.DCBiasIComponent == record.DCBiasQComponent)
106
107        ##Check against byte offset
108        assert( fid.tell() == sum([self.description.RecordLength,
109                                   self.description.DatasetSummaryRecordLength]))
110
111
112        #Return the validated record
113        return record

115    def parsePlatformPosition(self, fid):
116        '''
117        Parse the platform position record.
118        '''
119        from .PlatformPositionDataRecordType import (PlatformPositionDataRecordHeaderType,

123        ##Simple container
124        class Container(object):
125            pass
126
127        record = Container()
128
129        #First ensure that spec saus there is only one record
130        assert(self.description.NumberPlatformPositionRecords == 1)
131
132        #Start with the header of the record
133        header = PlatformPositionDataRecordHeaderType()
134        
135        #Read the header
136        header.fromfile(fid)
137
138        #ALOS specific validators
139        #http://www.ga.gov.au/__data/assets/pdf_file/0019/11719/GA10287.pdf Pg: 3-39
140        assert(header.RecordSequenceNumber == 3)
141        assert(header.FirstRecordType == 18)
142        assert(header.RecordTypeCode == 30)
143        assert(header.SecondRecordSubType == 18)
144        assert(header.ThirdRecordSubType == 20)
145        assert(header.RecordLength == 4680)
146        assert(header.OrbitalElementsDesignator in ['0', '1', '2'])
147        assert(header.NumberOfDataPoints == 28)
148        assert(header.TimeIntervalBetweenDataPointsInSec == 60.0)
149        record.header = header
150
151        #Now parse the state vectors individually
152        svs = []
153        for ii in range(header.NumberOfDataPoints):
154            sv = PlatformPositionDataRecordStateVectorType()
155            sv.fromfile(fid)
156            svs.append(sv)
157
158        record.statevectors = svs
159
160        #ALOS specific custom trailer
161        trailer = PlatformPositionDataRecordTrailerType()
162        trailer.fromfile(fid)
163        assert(trailer.LeapSecondFlag in [0,1])
164
165        #Check against byte offset
166        assert(fid.tell() == sum([self.description.RecordLength,
167                                  self.description.DatasetSummaryRecordLength,
168                                  self.description.PlatformPositionRecordLength]))
169
170        return record
