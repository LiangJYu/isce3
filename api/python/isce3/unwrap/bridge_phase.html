 15def bridge_unwrapped_phase(unw_phase: np.ndarray,
 16                           radius: int,
 17                           min_num_pixel: int,
 18                           erosion_size: int,
 19                           ramp_type: str,
 20                           deramp_max_num_sample: int) -> np.ndarray:
 21    """Bridge disconnected unwrapped phase regions by estimating and removing
 22    their relative phase offsets, which is interger of 2 pi.
 23
 24    In unwrapping, isolated components may be unwrapped independently,
 25    producing integer cycle jumps between them. This function finds
 26    nearby components (within `radius` pixels), computes the median phase
 27    difference at their nearest boundary pixels, shifts one region to align it,
 28    and merges the labels—effectively “bridging” the jump.
 29    Finally, it filters out speckle (small islands) and smooths
 30    mask edges via erosion.
 31
 32    Parameters
 33    ----------
 34    unw_phase : numpy.ndarray
 35        2D array representing the unwrapped phase image.
 36    radius : int
 37        The maximum radius used when bridging disconnected regions.
 38    min_num_pixel : int
 39        The minimum number of pixel of connected components to retain during processing.
 40    erosion_size : int
 41        The size of the structuring element used for erosion during labeling.
 42    ramp_type : str, optional
 43        Type of ramp to be estimated. Default is 'linear'.
 44        Possible options -
 45        'linear', 'quadratic', 'linear_range', 'linear_azimuth'
 46        'quadratic_range', 'quadratic_azimuth'
 47    deramp_max_num_sample : float, optional
 48        Maximum number of pixel samples, above which uniform sampling is
 49        applied to reduce sample size. Default is 1e6.    Returns
 50    -------
 51    bridge_unw : numpy.ndarray
 52        2D array with the bridged unwrapped phase image.
 53    """
 54    channel = journal.info("isce3.unwrap.bridge_phase.bridge_unwrapped_phase")
 55
 56    runw_img_bool = unw_phase != 0
 57    label_img, num_cluster = nd_label(runw_img_bool, structure=np.ones((3, 3)))
 58    channel.log(f"Bridge algorithm : {num_cluster} clusters")
 59
 60    if num_cluster <= 1:
 61        channel.log(f"Bridge algorithm is not applied since all components are connected.")
 62        return unw_phase
 63
 64    channel.log(f"   radius: {radius}   min_num_pixel: {min_num_pixel}  erosion_size: {erosion_size} ")
 65    cc = bridgeConnectComponent(conncomp=label_img)
 66    cc.label(min_num_pixel=min_num_pixel, erosion_size=erosion_size)
 67    if cc.label_ref is not None:
 68        cc.find_mst_bridge()
 69        bridge_unw = cc.unwrap_conn_comp(
 70            unw_phase,
 71            radius=radius,
 72            ramp_type=ramp_type,
 73            max_num_sample=deramp_max_num_sample
 74            )
 75    else:
 76        bridge_unw = unw_phase
 77    return bridge_unw

 80def label_boundary(
 81        label_img: np.ndarray,
 82        num_label: int,
 83        erosion_size: int = 5,
 84        ) -> Tuple[np.ndarray, int, np.ndarray]:
 85    """
 86    Label the boundary of the labeled array.
 87
 88    This function labels the boundaries of connected components in a labeled
 89    array. It optionally applies morphological erosion to the labeled array
 90    before finding and labeling the boundaries.
 91
 92    Parameters
 93    ----------
 94    label_img : numpy.ndarray
 95        2D array of labeled regions.
 96    num_label : int
 97        The number of labels in the labeled array.
 98    erosion_size : int, optional
 99        The size of the structuring element used for morphological erosion.
100        Default is 5.
101
102    Returns
103    -------
104    label_img : numpy.ndarray
105        2D array with relabeled regions after erosion and boundary labeling.
106    num_label : int
107        The updated number of labels after relabeling.
108    label_bound : numpy.ndarray
109        2D array of labeled boundaries of the regions.
110    """
111    channel = journal.info("isce3.unwrap.bridge_phase.label_boundary")
112
113    if erosion_size > 0:
114        erosion_yy, erosion_xx = np.ogrid[-erosion_size:erosion_size + 1,
115                                          -erosion_size:erosion_size + 1]
116        # circle mask
117        erosion_structure = (erosion_xx ** 2 +
118                             erosion_yy ** 2) <= (erosion_size**2)
119        label_erosion_img = binary_erosion(
120            label_img,
121            structure=erosion_structure).astype(np.uint8)
122
123        labeled_array, _ = nd_label(label_erosion_img)
124        regions = find_objects(labeled_array)
125
126        if len(regions) < num_label:
127            channel.log(
128                "Regions lost during morphological erosion operation:")
129            erosion_labels = [label_erosion_img[region].max()
130                              for region in regions]
131            for i in range(1, num_label + 1):
132                if i not in erosion_labels:
133                    label_img[label_img == i] = 0
134
135    else:
136        label_erosion_img = label_img > 0
137    label_img, num_label = nd_label(label_img, structure=np.ones((3, 3)))
138    # Create a boundary map using binary dilation and subtracting the original image
139    boundary_img = binary_dilation(label_erosion_img) & ~label_erosion_img
140    label_bound = boundary_img.astype(np.uint8)
141    label_bound *= label_erosion_img
142
143    return label_img, num_label, label_bound

146def label_conn_comp(
147        mask: np.ndarray,
148        min_num_pixel: float = 2.5e3,
149        erosion_size: int = 5,
150        ) -> Tuple[np.ndarray, int]:
151    """
152    Label and clean up the connected components mask.
153
154    This function labels the connected components in a binary mask,
155    removes small objects below a specified area, and optionally applies
156    morphological erosion to refine the labels.
157
158    Parameters
159    ----------
160    mask : numpy.ndarray
161        2D binary array representing the mask of connected components.
162    min_num_pixel : float, optional
163        Minimum number of pixels for a region to be kept. Default is 2.5e3.
164    erosion_size : int, optional
165        Size of the structuring element used for morphological erosion.
166        Default is 5.
167
168    Returns
169    -------
170    label_img : numpy.ndarray
171        2D array with labeled connected components after cleaning and erosion.
172    num_label : int
173        The number of labels (connected components) after cleaning and erosion.
174    """
175    channel = journal.info("isce3.unwrap.bridge_phase.label_conn_comp")
176
177    # Label the connected components
178    label_img, num_label = nd_label(mask, structure=np.ones((3, 3)))
179
180    # Calculate min_num_pixel if not specified
181    channel.log(f"Removing regions with area < {int(min_num_pixel)}")
182
183    # Remove small objects
184    object_slices = find_objects(label_img)
185    for i, slice_ in enumerate(object_slices, start=1):
186        if slice_ is not None:
187            if np.sum(label_img[slice_] == i) < min_num_pixel:
188                label_img[label_img == i] = 0
189
190    # Re-label after removing small objects
191    label_img, num_label = nd_label(label_img, structure=np.ones((3, 3)))
192
193    # Apply morphological erosion if specified
194    if erosion_size > 0:
195        erosion_structure = np.ones((erosion_size, erosion_size), dtype=bool)
196        label_erosion_img = binary_erosion(
197            label_img > 0,
198            structure=erosion_structure).astype(np.uint8)
199
200        labeled_array, _ = nd_label(label_erosion_img)
201        regions = find_objects(labeled_array)
202
203        if len(regions) < num_label:
204            channel.log("Regions lost during morphological erosion operation:")
205            erosion_labels = [label_erosion_img[region].max()
206                              for region in regions]
207            for i in range(1, num_label + 1):
208                if i not in erosion_labels:
209                    label_img[label_img == i] = 0
210
211        # Re-label after erosion
212        label_img, num_label = nd_label(label_img, structure=np.ones((3, 3)))
213
214    return label_img, num_label

220    def __init__(self, conncomp: np.ndarray):
221        """Initialize the ConnectComponent object."""
222        if not isinstance(conncomp, np.ndarray):
223            raise ValueError("Input conncomp is not np.ndarray")
224        self.conncomp = conncomp
225        self.length, self.width = self.conncomp.shape

227    def label(self,
228              min_num_pixel: float = 2.5e3,
229              erosion_size: int = 5) -> None:
230        """
231        Label connected components in the image and identify the reference
232        label.
233
234        This function labels connected components in the input image based on a
235        minimum area threshold and performs boundary erosion. It also
236        identifies the reference label as the largest connected component.
237
238        Parameters
239        ----------
240        min_num_pixel : float, optional
241            Minimum area threshold for connected components. Default is 2500.
242        erosion_size : int, optional
243            Size of the structuring element for boundary erosion. Default is 5.
244
245        Returns
246        -------
247        None
248        """
249        channel = journal.info(
250            "isce3.unwrap.bridge_phase.bridgeConnectComponent")
251        self.labelImg, self.num_label = label_conn_comp(
252            self.conncomp, min_num_pixel=min_num_pixel)
253
254        if self.num_label == 1:
255            channel.log(f"Bridge algorithm is not applied because only one component exists.")
256        elif self.num_label == 0:
257            channel.log(f"Bridge algorithm is not applied because component does not exist.")
258
259        self.labelImg, self.num_label, self.labelBound = label_boundary(
260            self.labelImg,
261            self.num_label,
262            erosion_size=erosion_size
263            )
264
265        regions = find_objects(self.labelImg)
266        # if regions are not empty
267        if regions:
268            idx = np.argmax([np.sum(self.labelImg[region])
269                             for region in regions])
270            self.label_ref = idx + 1
271        # if regions are empty
272        else:
273            self.label_ref = None

275    def get_all_bridge(self) -> Tuple[Dict[str, Any], np.ndarray]:
276        """
277        Compute all possible bridges between labeled areas and their distances.
278
279        This function calculates the shortest distances between all pairs of
280        labeled areas in the label image using k-d trees for efficient
281        nearest-neighbor searches. It stores the connections and distances in
282        a dictionary and a distance matrix.
283
284        Parameters
285        ----------
286        None
287
288        Returns
289        -------
290        connDict : Dict[str, Any]
291            A dictionary containing the coordinates of the closest points
292            between each pair of labeled areas and their distances. Keys
293            are formatted as "{label1}_{label2}" with values as dictionaries
294            containing:
295            - label1: coordinates of the closest point in the first labeled
296                      area.
297            - label2: coordinates of the closest point in the second labeled
298                      area.
299            - "distance": the distance between these points.
300        distMat : np.ndarray
301            A symmetric matrix of shape (num_label, num_label) containing the
302            minimum distances between each pair of labeled areas.
303        """
304        self.connDict = {}
305        self.distMat = np.zeros((self.num_label, self.num_label),
306                                dtype=np.float32)
307        # if the number of labels is zero, then return empty dictionary and
308        # zero array
309        if self.num_label == 0:
310            return self.connDict, self.distMat
311
312        trees = [cKDTree(np.argwhere(self.labelImg == i + 1))
313                 for i in range(self.num_label)]
314
315        for i, j in itertools.combinations(range(self.num_label), 2):
316            dist, idx = trees[i].query(trees[j].data)
317            idx_min = np.argmin(dist)
318            yxi = trees[i].data[idx[idx_min]]
319            yxj = trees[j].data[idx_min]
320            dist_min = dist[idx_min]
321            n0, n1 = str(i + 1), str(j + 1)
322            conn = {n0: yxi, n1: yxj, "distance": dist_min}
323            self.connDict[f"{n0}_{n1}"] = conn
324            self.distMat[i, j] = self.distMat[j, i] = dist_min
325
326        return self.connDict, self.distMat

328    def find_mst_bridge(self) -> List[Dict[str, Union[int, float]]]:
329        """
330        Search for bridges to connect all labeled areas using the minimum
331        spanning tree algorithm.
332
333        This function finds the minimum set of connections (bridges) needed to
334        connect all labeled areas using the minimum spanning tree (MST)
335        algorithm. If the distance matrix (`self.distMat`) is not
336        available, it is computed using the `get_all_bridge` method.
337
338        Parameters
339        ----------
340        None
341
342        Returns
343        -------
344        bridges : List[Dict[str, Union[int, float]]]
345            A list of dictionaries, each representing a bridge with the
346            following keys:
347            - 'first_endpoint_x': x-coordinate of the first node.
348            - 'first_endpoint_y': y-coordinate of the first node.
349            - 'second_endpoint_x': x-coordinate of the second node.
350            - 'second_endpoint_y': y-coordinate of the second node.
351            - 'label0': label of the first node.
352            - 'label1': label of the second node.
353            - 'distance': Euclidean distance between the two nodes.
354        """
355        if not hasattr(self, "distMat"):
356            self.get_all_bridge()
357
358        distMatMst = csg.minimum_spanning_tree(self.distMat)
359        succs, preds = csg.breadth_first_order(
360            distMatMst, i_start=self.label_ref - 1, directed=False
361        )
362        self.bridges = []
363        for i in range(1, succs.size):
364            n0 = preds[succs[i]] + 1
365            n1 = succs[i] + 1
366            if n0 > n1:
367                nn = [str(n1), str(n0)]
368            else:
369                nn = [str(n0), str(n1)]
370            conn = self.connDict[f"{nn[0]}_{nn[1]}"]
371            first_endpoint_y, first_endpoint_x = conn[str(n0)]
372            second_endpoint_y, second_endpoint_x = conn[str(n1)]
373            bridge = dict()
374            bridge = {
375                "first_endpoint_x": int(first_endpoint_x),
376                "first_endpoint_y": int(first_endpoint_y),
377                "second_endpoint_x": int(second_endpoint_x),
378                "second_endpoint_y": int(second_endpoint_y),
379                "distance": float(
380                    ((second_endpoint_x - first_endpoint_x)**2 +
381                     (second_endpoint_y - first_endpoint_y)**2)**0.5),
382                "label0": n0,
383                "label1": n1,
384            }
385            self.bridges.append(bridge)
386        self.num_bridge = len(self.bridges)
387        return self.bridges

389    def get_bridge_endpoint_aoi_mask(
390        self, bridge: Dict[str, int], radius: int = 50
391    ) -> Tuple[np.ndarray, np.ndarray]:
392        """
393        Get the Area of Interest (AOI) mask for bridge endpoints.
394
395        This function generates AOI masks around the endpoints of a bridge
396        between two connected components in the phase data. The AOI masks are
397        square regions centered at the bridge endpoints with a specified
398        radius.
399
400        Parameters
401        ----------
402        bridge : Dict[str, int]
403            Dictionary containing the coordinates of the bridge endpoints.
404            Keys should include:
405            - "first_endpoint_x": x-coordinate of the first endpoint.
406            - "first_endpoint_y": y-coordinate of the first endpoint.
407            - "second_endpoint_x": x-coordinate of the second endpoint.
408            - "second_endpoint_y": y-coordinate of the second endpoint.
409        radius : int, optional
410            Radius of the square AOI around each bridge endpoint.
411            Default is 50.
412
413        Returns
414        -------
415        aoi_mask0 : np.ndarray
416            Boolean mask of the AOI around the first bridge endpoint.
417        aoi_mask1 : np.ndarray
418            Boolean mask of the AOI around the second bridge endpoint.
419        """
420        # 1) unpack the two endpoints
421        first_endpoint_x = bridge["first_endpoint_x"]
422        first_endpoint_y = bridge["first_endpoint_y"]
423        second_endpoint_x = bridge["second_endpoint_x"]
424        second_endpoint_y = bridge["second_endpoint_y"]
425
426        # 2) compute row/col bounds for the first endpoint
427        first_col_min = max(0, first_endpoint_x - radius)
428        first_col_max = min(self.width, first_endpoint_x  + radius + 1)
429        first_row_min = max(0, first_endpoint_y - radius)
430        first_row_max = min(self.length, first_endpoint_y  + radius + 1)
431
432        # 3) compute row/col bounds for the second endpoint
433        second_col_min = max(0, second_endpoint_x - radius)
434        second_col_max = min(self.width, second_endpoint_x + radius + 1)
435        second_row_min = max(0, second_endpoint_y - radius)
436        second_row_max = min(self.length, second_endpoint_y + radius + 1)
437
438        # 4) build the two AOI masks
439        aoi_mask_first = np.zeros(self.labelImg.shape, dtype=bool)
440        aoi_mask_second = np.zeros(self.labelImg.shape, dtype=bool)
441
442        aoi_mask_first[
443            first_row_min:first_row_max,
444            first_col_min:first_col_max
445        ] = True
446
447        aoi_mask_second[
448            second_row_min:second_row_max,
449            second_col_min:second_col_max
450        ] = True
451
452        return aoi_mask_first, aoi_mask_second

454    def unwrap_conn_comp(
455        self,
456        unw: np.ndarray,
457        radius: int = 50,
458        ramp_type: Optional[str] = None,
459        max_num_sample: int = 1e6,
460    ) -> np.ndarray:
461        """
462        Perform bridging to unwrap connected components in the phase data.
463
464        This function unrolls the phase data by bridging the gaps between
465        connected components, optionally removing any ramp present in the data.
466
467        Parameters
468        ----------
469        unw : np.ndarray
470            2D array of unwrapped phase data.
471        radius : int, optional
472            Radius of the area of interest (AOI) around bridge endpoints for
473            unwrapping. Default is 50.
474        ramp_type : Optional[str], optional
475            Type of ramp to be estimated and removed before unwrapping.
476            If None, no ramp is removed. Default is None.
477
478        Returns
479        -------
480        unw : np.ndarray
481            2D array of unwrapped phase data after bridging.
482        """
483        channel = journal.info(
484            "isce3.unwrap.bridge_phase.bridgeConnectComponent")
485        radius = int(min(radius, min(self.conncomp.shape) * 0.5))
486        unw = np.array(unw, dtype=np.float32)
487
488        if ramp_type is not None:
489            channel.log(f"Estimating a {ramp_type} ramp")
490            ramp_mask = self.labelImg == self.label_ref
491            unw, ramp = deramp(unw, ramp_mask, ramp_type, max_num_sample)
492
493        for bridge in self.bridges:
494            aoi_mask0, aoi_mask1 = self.get_bridge_endpoint_aoi_mask(
495                bridge, radius=radius
496            )
497            label_mask0 = self.labelImg == bridge["label0"]
498            label_mask1 = self.labelImg == bridge["label1"]
499            value0 = np.nanmedian(unw[np.logical_and(aoi_mask0, label_mask0)])
500            value1 = np.nanmedian(unw[np.logical_and(aoi_mask1, label_mask1)])
501            diff_value = value1 - value0
502            num_jump = (np.abs(diff_value) + np.pi) // (2.0 * np.pi)
503            if diff_value > 0:
504                num_jump *= -1
505            unw[label_mask1] += 2.0 * np.pi * num_jump
506
507        if ramp_type is not None:
508            unw += ramp
509
510        return unw

513def deramp(data,
514           mask_in=None,
515           ramp_type='linear',
516           max_num_sample=1e6,
517           ignore_zero_value=True):
518    """Remove ramp from input data matrix based on pixel marked by
519    mask. Ignore data with NaN or zero value.
520
521    Parameters
522    ----------
523    data : np.ndarray
524        2D or 3D array of data to be deramped. If 3D, it's in the size of
525        (num_date, length, width).
526    mask_in : np.ndarray, optional
527        2D array mask of pixels used for ramp estimation.
528    ramp_type : str, optional
529        Type of ramp to be estimated. Default is 'linear'.
530        Possible options -
531        'linear', 'quadratic', 'linear_range', 'linear_azimuth'
532        'quadratic_range', 'quadratic_azimuth'
533    max_num_sample : float, optional
534        Maximum number of pixel samples, above which uniform sampling is
535        applied to reduce sample size. Default is 1e6.
536    ignore_zero_value : bool, optional
537        Ignore pixels with zero values. Default is True.
538        Recommended: True for phase data and False for offset data.
539
540    Returns
541    -------
542    data_out : np.ndarray
543        2D or 3D array of data after deramping.
544    ramp : np.ndarray
545        2D or 3D array of the estimated ramp.
546    """
547    dshape = data.shape
548    length, width = dshape[-2:]
549    num_pixel = length * width
550
551    # prepare input data
552    if len(dshape) == 3:
553        data = np.moveaxis(data, 0, -1)
554        data = data.reshape(num_pixel, -1)
555        dmean = np.mean(data, axis=-1).flatten()
556    else:
557        data = data.reshape(-1, 1)
558        dmean = np.array(data).flatten()
559
560    # mask
561    # 1. default
562    if mask_in is None:
563        mask_in = np.ones((length, width), dtype=np.float32)
564    mask = (mask_in != 0).flatten()
565    del mask_in
566
567    # 2. ignore pixels with NaN and/or zero data value
568    mask *= ~np.isnan(dmean)
569    if ignore_zero_value:
570        mask *= dmean != 0.
571    del dmean
572
573    # 3. for big dataset: uniformally sample the data for ramp estimation
574    mask_sum = np.sum(mask) 
575    if max_num_sample and mask_sum > max_num_sample:
576        step = int(np.ceil(np.sqrt(mask_sum / max_num_sample)))
577        if step > 1:
578            sample_flag = np.zeros((length, width), dtype=np.bool_)
579            sample_flag[int(step/2)::step,
580                        int(step/2)::step] = 1
581            mask *= sample_flag.flatten()
582            del sample_flag
583
584    # design matrix
585    xx, yy = np.meshgrid(np.arange(0, width),
586                         np.arange(0, length))
587    xx = np.array(xx, dtype=np.float32).reshape(-1, 1)
588    yy = np.array(yy, dtype=np.float32).reshape(-1, 1)
589    ones = np.ones(xx.shape, dtype=np.float32)
590    if ramp_type == 'linear':
591        G = np.hstack((yy, xx, ones))
592    elif ramp_type == 'quadratic':
593        G = np.hstack((yy**2, xx**2, yy*xx, yy, xx, ones))
594    elif ramp_type == 'linear_range':
595        G = np.hstack((xx, ones))
596    elif ramp_type == 'linear_azimuth':
597        G = np.hstack((yy, ones))
598    elif ramp_type == 'quadratic_range':
599        G = np.hstack((xx**2, xx, ones))
600    elif ramp_type == 'quadratic_azimuth':
601        G = np.hstack((yy**2, yy, ones))
602    else:
603        raise ValueError(f'un-recognized ramp type: {ramp_type}')
604
605    # estimate ramp
606    X = np.dot(np.linalg.pinv(G[mask, :], rcond=1e-15), data[mask, :])
607    ramp = np.dot(G, X)
608    ramp = np.array(ramp, dtype=data.dtype)
609
610    # do not change pixel with original zero value
611    if ignore_zero_value:
612        ramp[data == 0] = 0
613
614    data_out = data - ramp
615    if len(dshape) == 3:
616        ramp = np.moveaxis(ramp, -1, 0)
617        data_out = np.moveaxis(data_out, -1, 0)
618    ramp = ramp.reshape(dshape)
619    data_out = data_out.reshape(dshape)
620    return data_out, ramp

521    mask. Ignore data with NaN or zero value.
522
523    Parameters
524    ----------
525    data : np.ndarray
526        2D or 3D array of data to be deramped. If 3D, it's in the size of
527        (num_date, length, width).
528    mask_in : np.ndarray, optional
529        2D array mask of pixels used for ramp estimation.
530    ramp_type : str, optional
531        Type of ramp to be estimated. Default is 'linear'.
532        Possible options -
533        'linear', 'quadratic', 'linear_range', 'linear_azimuth'
534        'quadratic_range', 'quadratic_azimuth'
535    max_num_sample : float, optional
536        Maximum number of pixel samples, above which uniform sampling is
537        applied to reduce sample size. Default is 1e6.
538    ignore_zero_value : bool, optional
539        Ignore pixels with zero values. Default is True.
540        Recommended: True for phase data and False for offset data.
541
542    Returns
543    -------
544    data_out : np.ndarray
545        2D or 3D array of data after deramping.
546    ramp : np.ndarray
547        2D or 3D array of the estimated ramp.
548    """
549    dshape = data.shape
550    length, width = dshape[-2:]
551    num_pixel = length * width
552
553    # prepare input data
554    if len(dshape) == 3:
555        data = np.moveaxis(data, 0, -1)
556        data = data.reshape(num_pixel, -1)
557        dmean = np.mean(data, axis=-1).flatten()
558    else:
559        data = data.reshape(-1, 1)
560        dmean = np.array(data).flatten()
561
562    # mask
563    # 1. default
564    if mask_in is None:
565        mask_in = np.ones((length, width), dtype=np.float32)
566    mask = (mask_in != 0).flatten()
567    del mask_in
568
569    # 2. ignore pixels with NaN and/or zero data value
570    mask *= ~np.isnan(dmean)
571    if ignore_zero_value:
572        mask *= dmean != 0.
573    del dmean
574

576    mask_sum = np.sum(mask) 
577    if max_num_sample and mask_sum > max_num_sample:
578        step = int(np.ceil(np.sqrt(mask_sum / max_num_sample)))
579        if step > 1:
580            sample_flag = np.zeros((length, width), dtype=np.bool_)
581            sample_flag[int(step/2)::step,
582                        int(step/2)::step] = 1
583            mask *= sample_flag.flatten()
584            del sample_flag
585

587    xx, yy = np.meshgrid(np.arange(0, width),
588                         np.arange(0, length))
589    xx = np.array(xx, dtype=np.float32).reshape(-1, 1)
590    yy = np.array(yy, dtype=np.float32).reshape(-1, 1)
591    ones = np.ones(xx.shape, dtype=np.float32)
592    if ramp_type == 'linear':
593        G = np.hstack((yy, xx, ones))
594    elif ramp_type == 'quadratic':
595        G = np.hstack((yy**2, xx**2, yy*xx, yy, xx, ones))
596    elif ramp_type == 'linear_range':
597        G = np.hstack((xx, ones))
598    elif ramp_type == 'linear_azimuth':
599        G = np.hstack((yy, ones))
600    elif ramp_type == 'quadratic_range':

602    elif ramp_type == 'quadratic_azimuth':

604    else:

611

614        ramp[data == 0] = 0
615

620    ramp = ramp.reshape(dshape)
621    data_out = data_out.reshape(dshape)
622    return data_out, ramp
