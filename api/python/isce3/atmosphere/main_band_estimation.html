 39    def compute_disp_nondisp(
 40            self,
 41            phi_sub_low=None,
 42            phi_sub_high=None,
 43            phi_diff_low_high=None,
 44            phi_main=None,
 45            phi_side=None,
 46            phi_diff_ms=None,
 47            slant_main=None,
 48            slant_side=None,
 49            comm_unwcor_coef=None,
 50            diff_unwcor_coef=None,
 51            no_data=0):
 52
 53        """Estimates dispersive and non-dispersive phase using given
 54        spectral diversity method. Note that each methods require different
 55        unwrapped interferograms.
 56        - split_main_band requires [phi_sub_low, phi_sub_high]
 57        - main_side_band requires [phi_main, phi_side]
 58        - main_diff_ms_band requires [phi_main, phi_side]
 59        If unwrapping correction terms are given, unwrapped phase array
 60        are corrected.
 61
 62        Parameters
 63        ----------
 64        phi_sub_low : numpy.ndarray
 65            unwrapped phase array of low sub-band interferogram
 66        phi_sub_high : numpy.ndarray
 67            unwrapped phase array of high sub-band interferogram
 68        phi_diff_low_high : numpy.ndarray
 69            unwrapped phase array of difference between low and
 70            high sub-band interferograms
 71        phi_main : numpy.ndarray
 72            unwrapped phase array of frequency A interferogram
 73        phi_side : numpy.ndarray
 74            unwrapped phase array of frequency B interferogram
 75        slant_main : numpy.ndarray
 76            slant range array of frequency A interferogram
 77        slant_side : numpy.ndarray
 78            slant range array of frequency B interferogram
 79        comm_unwcor_coef : numpy.ndarray
 80            common correction coefficient of unwrapped phases
 81        diff_unwcor_coef : numpy.ndarray
 82            differential correction coefficient of unwrapped phases
 83        no_data : float
 84            no data value
 85
 86        Returns
 87        -------
 88        dispersive : numpy.ndarray
 89            numpy array of dispersive array
 90        non_dispersive : numpy.ndarray
 91            non-dispersive phase array
 92        """
 93        error_channel = journal.error(
 94            'MainBandIonosphereEstimation.compute_disp_nondisp')
 95
 96        # Validate mandatory inputs for main/side-band or main/diff-MS
 97        if phi_main is None or phi_side is None:
 98            err_str = "unwrapped interferogram array main and side band"\
 99                "is required."
100            error_channel.log(err_str)
101            raise ValueError(err_str)
102        # Match spatial sampling by decimating main-band to side-band spacing
103        phi_main = decimate_freq_a_array(slant_main,
104                                         slant_side,
105                                         phi_main)
106        if phi_diff_ms is not None:
107            no_data_array = (phi_main == no_data) |\
108                            (phi_diff_ms == no_data)
109        else:
110            no_data_array = (phi_main == no_data) |\
111                            (phi_side == no_data)
112
113        # correct unwrapped phase, if coefficients are supplied
114        if comm_unwcor_coef is not None and diff_unwcor_coef is not None:
115            if phi_diff_ms is not None:
116                phi_main = phi_main - 2 * np.pi * comm_unwcor_coef
117                phi_diff_ms = phi_diff_ms - 2 * np.pi * diff_unwcor_coef
118            else:
119                phi_main = phi_main - 2 * np.pi * comm_unwcor_coef
120                phi_side = phi_side - 2 * np.pi *\
121                    (comm_unwcor_coef + diff_unwcor_coef)
122        # Estimate dispersive / non-dispersive components after correcting
123        # unwrapped phase
124        dispersive, non_dispersive = self.estimate_iono(
125            f0=self.f0,
126            f1=self.f1,
127            phi0=phi_main,
128            phi1=phi_side,
129            phi_diff_ms=phi_diff_ms)
130
131        dispersive = np.where(no_data_array, no_data, dispersive)
132        non_dispersive = np.where(no_data_array, no_data, non_dispersive)
133        return dispersive, non_dispersive

259        diff_ms_array : numpy.ndarray

370    def estimate_iono_std(
371            self,
372            main_coh=None,
373            side_coh=None,
374            low_band_coh=None,
375            high_band_coh=None,
376            diff_low_high_coh=None,
377            slant_main=None,
378            slant_side=None,
379            number_looks=1,
380            resample_flag=True):
381        """Calculate the theoretical standard deviation of
382        the ionospheric phase based on the coherencess
383
384        Parameters
385        ----------
386        main_coh : numpy.ndarray
387            coherence of main-band interferogram
388        side_coh : numpy.ndarray
389            coherence of side-band interferogram
390        low_band_coh : numpy.ndarray
391            coherence of low subband interferogram
392        high_band_coh : numpy.ndarray
393            coherence of high subband interferogram
394        diff_low_high_band_array : numpy.ndarray
395            coherence of difference (high-low) interferogram
396        slant_main : numpy.ndarray
397            slant range array of frequency A band
398        slant_side : numpy.ndarray
399            slant range array of frequency B band
400        number_looks : int
401            number of looks
402        resample_flag : bool
403
404        Returns
405        -------
406        sig_phi_iono : numpy.ndarray
407            phase standard deviation of ionosphere phase
408        sig_nondisp : numpy.ndarray
409            phase standard deviation of non-dispersive
410        """
411        # resample coherences array of frequency A to
412        # frequency B grid
413        if (side_coh is not None) and (resample_flag):
414            if slant_main is None:
415                slant_main = self.slant_main
416            if slant_side is None:
417                slant_side = self.slant_side
418
419            main_coh = decimate_freq_a_array(
420                slant_main,
421                slant_side,
422                main_coh)
423
424        # estimate sigma from main- and side- band coherences
425        if (main_coh is not None) & (side_coh is not None):
426            sig_phi_main = np.divide(
427                np.sqrt(1 - main_coh**2),
428                main_coh / np.sqrt(2 * number_looks),
429                out=np.zeros_like(main_coh),
430                where=main_coh != 0)
431            sig_phi_side = np.divide(
432                np.sqrt(1 - side_coh**2),
433                side_coh / np.sqrt(2 * number_looks),
434                out=np.zeros_like(side_coh),
435                where=side_coh != 0)
436
437        sig_phi_iono, sig_nondisp = \
438            self.estimate_sigma(
439                sig_phi_main,
440                sig_phi_side)
441
442        return sig_phi_iono, sig_nondisp

444    def estimate_sigma_main_diff(
445            self,
446            sig_phi0,
447            sig_phi1):
448        """Estimate sigma from coherence for main_diff_ms method
449
450        Parameters
451        ----------
452        sig_phi0 : numpy.ndarray
453            phase standard deviation of main-band interferogram
454        sig_phi1 : numpy.ndarray
455            phase standard deviation of side-band interferogram
456
457        Returns
458        -------
459        sig_iono : numpy.ndarray
460            2D array of phase standard deviation of dispersive
461        sig_nondisp : numpy.ndarray
462            2D array of phase standard deviation of non-dispersive
463        """
464
465        a = self.f1 / (self.f1 + self.f0)
466        b = (self.f0 * self.f1) / (self.f0**2 - self.f1**2)
467        sig_phi01 = np.sqrt(sig_phi0**2 + sig_phi1**2)
468        sig_iono = np.sqrt(a**2 * sig_phi0**2 + b**2 * sig_phi01**2)
469
470        c = self.f0**2 / (self.f0**2 - self.f1**2)
471        d = self.f0 * self.f1 / (self.f0**2 - self.f1**2)
472        sig_nondisp = np.sqrt(c**2 * sig_phi0**2 + d**2 * sig_phi01**2)
473
474        return sig_iono, sig_nondisp

476    def estimate_sigma_main_side(
477            self,
478            sig_phi0,
479            sig_phi1):
480        """Estimate sigma from coherence for main-side method
481
482        Parameters
483        ----------
484        sig_phi0 : numpy.ndarray
485            phase standard deviation of main-band interferogram
486        sig_phi1 : numpy.ndarray
487            phase standard deviation of side-band interferogram
488
489        Returns
490        -------
491        sig_iono : numpy.ndarray
492            phase standard deviation of ionosphere phase
493        sig_nondisp : numpy.ndarray
494            2D array of phase standard deviation of non-dispersive
495        """
496        a = (self.f1**2) / (self.f1**2 - self.f0**2)
497        b = (self.f0 * self.f1) / (self.f1**2 - self.f0**2)
498        c = (self.f0) / (self.f1**2 - self.f0**2)
499
500        sig_iono = np.sqrt(a**2 * sig_phi0**2 + b**2 * sig_phi1**2)
501        sig_non_disp = np.sqrt(c**2 * (self.f0**2 * sig_phi0**2
502                               + self.f1**2 * sig_phi1**2))
503
504        return sig_iono, sig_non_disp

663        2D differential unwrapping error coefficient array
664    """
665    f_diff = f1 - f0
666
667    com_unw_coeff = np.round(
668        (main_runw - (nondisp_array + disp_array)) / (2 * np.pi))
669    term1 = f_diff / f0 * nondisp_array
670    term2 = f_diff / f1 * disp_array
671    diff_unw_coeff = np.round(
672        (diff_ms_runw + term1 - term2) / (2 * np.pi))
673
674    return com_unw_coeff, diff_unw_coeff

679        f1,

685        freq_low=None,
686        freq_high=None,

689        diff_low_high_runw=None):
690
691    """Compute unwrapping error coefficients.
692
693    Parameters
694    ----------
695    f0 : float
696        radar center frequency of frequency A band
697    f1 : float
698        radar center frequency of frequency B band
699    disp_array : numpy.ndarray
700        2D dispersive array estimated from given methods
701    nondisp_array : numpy.ndarray
702        2D non-dispersive array estimated from given methods
703    main_runw : numpy.ndarray
704        2D runw array of main-band interferogram
705    side_runw : numpy.ndarray
706        2D runw array of of side-band interferogram
707
708    Returns
709    -------
710    com_unw_coeff : numpy.ndarray
711        2D common unwrapping error coefficient array
712    diff_unw_coeff : numpy.ndarray
713        2D differential unwrapping error coefficient array
714    """
715
716    # Unwrapping errors exists in side interferogram,
717    # but not in main interferogram.
718    diff_unw_coeff = np.round(((1 - f1 / f0)
719        * nondisp_array + (1 - f0 / f1) * disp_array

725        - (1 + f0 / f1) * disp_array
726        - 2 * np.pi * diff_unw_coeff) / (4 * np.pi))
727

731def estimate_iono_main_diff(f0,
732                            f1,
733                            phi0,
734                            phi_diff_ms,
735                            phi1=None):
736
737    """Estimates ionospheric phase from main-band
738    and the difference of main and side-band interferograms
739
740    Parameters
741    ----------
742    f0 : float
743        radar center frequency of main band
744    f1 : float
745        radar center frequency of side band
746    phi0 : numpy.ndarray
747        numpy array of main interferogram
748    phi_diff_ms : numpy.ndarray
749        numpy array of difference between main and side interferogram
750
751    Returns
752    -------
753    dispersive : numpy.ndarray
754        numpy array of estimated dispersive
755    non_dispersive : numpy.ndarray
756        numpy array of estimated non-dispersive
757    """
758    y_size, x_size = phi0.shape
759    d = np.ones((2, y_size * x_size))
760    d[0, :] = phi0.flatten()
761    d[1, :] = phi_diff_ms.flatten()
762
763    # Coefficient matrix for solving dispersive and non-dispersive
764    # components
765    coeff_mat = np.ones((2, 2))
766    coeff_mat[1, 0] = (f1 - f0) / f1
767    coeff_mat[1, 1] = (f0 - f1) / f0
768    coeff_mat1 = np.linalg.pinv(coeff_mat)
769    output = np.dot(coeff_mat1, d)
770
771    non_dispersive = output[1].reshape(y_size, x_size)
772    dispersive = output[0].reshape(y_size, x_size)
773
774    return dispersive, non_dispersive

445    def estimate_sigma_main_diff(
446            self,
447            sig_phi0,
448            sig_phi1):
449        """Estimate sigma from coherence for main_diff_ms method
450
451        Parameters
452        ----------
453        sig_phi0 : numpy.ndarray
454            phase standard deviation of main-band interferogram
455        sig_phi1 : numpy.ndarray
456            phase standard deviation of side-band interferogram
457
458        Returns
459        -------
460        sig_iono : numpy.ndarray
461            2D array of phase standard deviation of dispersive
462        sig_nondisp : numpy.ndarray
463            2D array of phase standard deviation of non-dispersive
464        """
465
466        a = self.f1 / (self.f1 + self.f0)
467        b = (self.f0 * self.f1) / (self.f0**2 - self.f1**2)
468        sig_phi01 = np.sqrt(sig_phi0**2 + sig_phi1**2)
469        sig_iono = np.sqrt(a**2 * sig_phi0**2 + b**2 * sig_phi01**2)
470
471        c = self.f0**2 / (self.f0**2 - self.f1**2)
472        d = self.f0 * self.f1 / (self.f0**2 - self.f1**2)
473        sig_nondisp = np.sqrt(c**2 * sig_phi0**2 + d**2 * sig_phi01**2)
474
475        return sig_iono, sig_nondisp

478            self,
479            sig_phi0,
480            sig_phi1):
481        """Estimate sigma from coherence for main-side method
482
