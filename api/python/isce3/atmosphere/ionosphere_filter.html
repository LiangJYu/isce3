 59    def low_pass_filter(
 60            self,
 61            input_data,
 62            input_std_dev,
 63            mask_path,
 64            filtered_output,
 65            filtered_std_dev,
 66            lines_per_block,
 67            min_cluster_pixels):
 68        """Apply low_pass_filtering for dispersive and nondispersive
 69        with standard deviation. Before filtering, fill the gaps with
 70        smoothed or nearest values.
 71
 72        Parameters
 73        ----------
 74        input_data : str
 75            file path for data to be filtered.
 76        input_std_dev : str
 77            file path for standard deviation
 78            or nondispersive array
 79        mask_path : str
 80            file path for mask raster
 81            1: valid pixels,
 82            0: invalid pixels.
 83        filtered_output : str
 84            output file path or h5py dataset to write the filtered data
 85        filtered_std_dev : str
 86            output file path or h5py dataset to write filtered standard
 87            deviation.
 88
 89        Returns
 90        -------
 91        """
 92        data_shape, _ = get_raster_info(input_data)
 93        data_length, data_width = data_shape
 94        # Determine number of blocks to process
 95        lines_per_block = min(data_length,
 96                              lines_per_block)
 97        # Determine the amount of padding
 98        pad_length = 2 * (self.y_kernel // 2)
 99        pad_width = 2 * (self.x_kernel // 2)
100        pad_shape = (pad_length, pad_width)
101
102        # Prepare to write output to files
103        for output in [filtered_output, filtered_std_dev]:
104            if not isinstance(output, h5py.Dataset) and \
105               not os.path.isfile(output):
106                raster = isce3.io.Raster(
107                    path=output,
108                    width=data_width,
109                    length=data_length,
110                    num_bands=1,
111                    dtype=gdal.GDT_Float32,
112                    driver_name='ENVI')
113                del raster
114
115        for iter_cnt in range(self.iteration):
116
117            block_params = block_param_generator(
118                lines_per_block, data_shape, pad_shape)
119            # Start block processing
120            for block_param in block_params:
121                width_offset = pad_width // 2
122                length_offset = pad_length // 2
123
124                # Prepare to write temp_files
125                filtered_iono_temp_input_path = \
126                    f'{self.outputdir}/filtered_iono_temp{iter_cnt-1}'
127                filtered_std_temp_input_path = \
128                    f'{self.outputdir}/filtered_iono_std_temp{iter_cnt-1}'
129
130                block_data_path = filtered_iono_temp_input_path \
131                    if iter_cnt > 0 else input_data
132                data_block = read_block_array(block_data_path, block_param)
133                block_sig_path = filtered_std_temp_input_path \
134                    if iter_cnt > 0 else input_std_dev
135                data_sig_block = read_block_array(block_sig_path, block_param)
136                mask_block = read_block_array(mask_path, block_param,
137                                              fill_value=0)
138
139                # buffer region represents the additional buffer areas for
140                # the block.
141                buffer_region = np.ones_like(mask_block, dtype=bool)
142                buffer_region[
143                    length_offset:length_offset+block_param.block_length,
144                    width_offset:width_offset+block_param.data_width] = 0
145                mask0 = mask_block == 0
146                mask1 = mask0 == 0
147
148                data_block[mask0] = np.nan
149
150                # remove small areas from images to avoid
151                # the possibility of unwrap errors
152                data_block = remove_small_components(
153                    data_block,
154                    min_cluster_pixels=min_cluster_pixels)
155
156                if self.filling_method == "smoothed":
157                    fill_method = fill_with_smoothed
158                elif self.filling_method == "nearest":
159                    fill_method = fill_nearest
160                elif self.filling_method == "distance":
161                    fill_method = fill_with_distance
162
163                if self.filling_method in ["distance"]:
164                    weight = mask_block.astype('float')
165                    filled_data = fill_method(data_block, weight)
166                    filled_data_sig = fill_method(data_sig_block, weight)
167
168                else:
169                    filled_data_sig = fill_method(data_sig_block)
170                    filled_data = fill_method(data_block)
171
172                if iter_cnt > 0:
173                    # Replace the valid pixels with original unfiltered data
174                    # to avoid too much smoothed signal
175                    unfilt_data_block = read_block_array(input_data,
176                                                         block_param)
177                    filled_data[mask1] = unfilt_data_block[mask1]
178                    unfilt_data_block_sig = read_block_array(input_std_dev,
179                                                             block_param)
180                    filled_data_sig[mask1] = unfilt_data_block_sig[mask1]
181
182                # after filling gaps, filter the data
183                filt_data, filt_data_sig = filter_data_with_sig(
184                    input_array=filled_data,
185                    sig_array=filled_data_sig,
186                    kernel_width=self.x_kernel,
187                    kernel_length=self.y_kernel,
188                    sig_kernel_x=self.sig_x,
189                    sig_kernel_y=self.sig_y)
190
191                # set output to HDF5 for final iteration
192                # otherwise write to temp file
193                if iter_cnt == self.iteration - 1:
194                    output_iono = filtered_output
195                    output_std = filtered_std_dev
196                else:
197                    output_iono = \
198                        f'{self.outputdir}/filtered_iono_temp{iter_cnt}'
199                    output_std = \
200                        f'{self.outputdir}/filtered_iono_std_temp{iter_cnt}'
201
202                write_array(
203                    output_iono,
204                    filt_data,
205                    block_row=block_param.write_start_line,
206                    data_shape=data_shape)
207
208                write_array(
209                    output_std,
210                    filt_data_sig,
211                    block_row=block_param.write_start_line,
212                    data_shape=data_shape)

215def fill_with_smoothed(data):
216    """Replace the value of nan 'data' cells
217    by the value of the linear interpolated data cell.
218    The values, not covered by interpolation, are filled
219    with nearest values.
220
221    Parameters
222    ----------
223    data : numpy.ndarray
224        array containing holes to be filled.
225        nan values are considered as holes.
226
227    Returns
228    -------
229    numpy.ndarray
230        array with no data values filled with data values
231        from numpy.griddata
232    """
233    rows, cols = data.shape
234    x = np.arange(0, cols)
235    y = np.arange(0, rows)
236    xx, yy = np.meshgrid(x, y)
237
238    xx = xx.ravel()
239    yy = yy.ravel()
240    data = data.ravel()
241
242    is_nan_mask = np.isnan(data)
243    not_nan_mask = np.invert(is_nan_mask)
244
245    if np.all(not_nan_mask):
246        return data.reshape([rows, cols])
247
248    # find x and y where valid values are located.
249    xx_wo_nan = xx[not_nan_mask]
250    yy_wo_nan = yy[not_nan_mask]
251    data_wo_nan = data[not_nan_mask]
252
253    xnew = xx[np.isnan(data)]
254    ynew = yy[np.isnan(data)]
255
256    # linear interpolation with griddata
257    znew = griddata((xx_wo_nan, yy_wo_nan),
258                    data_wo_nan,
259                    (xnew, ynew),
260                    method='linear')
261    data_filt = data.copy()
262    data_filt[np.isnan(data)] = znew
263    n_nonzero = np.sum(np.count_nonzero(np.isnan(data_filt)))
264
265    if n_nonzero > 0:
266        idx2 = np.isnan(data_filt)
267
268        xx_wo_nan = xx[np.invert(idx2)]
269        yy_wo_nan = yy[np.invert(idx2)]
270        data_wo_nan = data_filt[np.invert(idx2)]
271        xnew = xx[idx2]
272        ynew = yy[idx2]
273
274        # extrapolation using nearest values
275        znew_ext = griddata((xx_wo_nan, yy_wo_nan),
276                            data_wo_nan,
277                            (xnew, ynew),
278                            method='nearest')
279        data_filt[np.isnan(data_filt)] = znew_ext
280    return data_filt.reshape([rows, cols])

283def filter_data_with_sig(
284        input_array,
285        sig_array,
286        kernel_width,
287        kernel_length,
288        sig_kernel_x,
289        sig_kernel_y,
290        mask_array=None):
291    """ Filter input array by applying weighting
292    based on the statndard deviations
293    Parameters
294    ----------
295    input_array : numpy.ndarray
296        2D dispersive or nondispersive array
297    sig_array : numpy.ndarray
298        2D standard deviation array of dispersive
299        or nondispersive array
300    kernel_width : int
301        x kernel size for gaussian filtering
302    kernel_length : int
303        y kernel size for gaussian filtering
304    sig_kernel_x : int
305        x standard deviation for gaussian window
306    sig_kernel_y : int
307        y standard deviation for gaussian window
308
309    Returns
310    -------
311    filt_data : numpy.ndarray
312        2D filtered image
313    filt_data_sig : numpy.ndarray
314        2D filtered standard deviation image
315    """
316    # Create Gaussian kernel for filtering
317    kernel_rows = create_gaussian_kernel(kernel_length, sig_kernel_y)
318    kernel_rows = np.reshape(kernel_rows, (len(kernel_rows), 1))
319    kernel_cols = create_gaussian_kernel(kernel_width, sig_kernel_x)
320    kernel_cols = np.reshape(kernel_cols, (1, len(kernel_cols)))
321
322    sig_array_sqr = sig_array ** 2
323    input_div_sig = np.divide(
324        input_array,
325        sig_array_sqr,
326        out=np.zeros_like(input_array),
327        where=sig_array_sqr != 0)
328
329    inv_sig = np.divide(
330        1,
331        sig_array_sqr,
332        out=np.zeros_like(sig_array_sqr),
333        where=sig_array_sqr != 0)
334
335    if mask_array is not None:
336        filt_input_div_sig = isce3.signal.convolve2D(
337            input_div_sig,
338            mask_array,
339            kernel_cols,
340            kernel_rows,
341            False)
342
343        filt_inv_sig = isce3.signal.convolve2D(
344            inv_sig,
345            mask_array,
346            kernel_cols,
347            kernel_rows,
348            False)
349
350        filt_inv_sig_kernel2 = isce3.signal.convolve2D(
351            inv_sig,
352            mask_array,
353            kernel_cols**2,
354            kernel_rows**2,
355            False)
356
357    else:
358        filt_input_div_sig = isce3.signal.convolve2D(
359            input_div_sig,
360            kernel_cols,
361            kernel_rows,
362            False)
363
364        filt_inv_sig = isce3.signal.convolve2D(
365            inv_sig,
366            kernel_cols,
367            kernel_rows,
368            False)
369
370        filt_inv_sig_kernel2 = isce3.signal.convolve2D(
371            inv_sig,
372            kernel_cols**2,
373            kernel_rows**2,
374            False)
375
376    filt_data = np.divide(filt_input_div_sig,
377                          filt_inv_sig,
378                          out=np.zeros_like(filt_input_div_sig),
379                          where=filt_inv_sig != 0)
380
381    filt_data_sig = np.divide(filt_inv_sig_kernel2,
382                              filt_inv_sig**2,
383                              out=np.zeros_like(filt_inv_sig_kernel2),
384                              where=filt_inv_sig != 0)
385
386    filt_data_sig = np.sqrt(filt_data_sig)
387
388    return filt_data, filt_data_sig

391def read_block_array(raster, block_param, fill_value=0):
392    ''' Get a block of data from raster.
393        Raster can be a HDF5 file or a GDAL-friendly raster
394
395    Parameters
396    ----------
397    raster: h5py.Dataset or str
398        Raster where a block is to be read from. String value represents a
399        filepath for GDAL rasters.
400    block_param: BlockParam
401        Object specifying size of block and where to read from raster,
402        and amount of padding for the read array
403    fill_value: float
404        Pads with a fill value.
405    Returns
406    -------
407    data_block: np.ndarray
408        Block read from raster with shape specified in block_param.
409    '''
410    if isinstance(raster, h5py.Dataset):
411        data_block = np.empty((block_param.read_length,
412                               block_param.data_width),
413                              dtype=raster.dtype)
414        raster.read_direct(
415            data_block,
416            np.s_[block_param.read_start_line:
417                  block_param.read_start_line + block_param.read_length,
418                  0: block_param.data_width])
419    else:
420        # Open input data using GDAL to get raster length
421        ds_data = gdal.Open(raster, gdal.GA_Update)
422        data_block = ds_data.GetRasterBand(1).ReadAsArray(
423            0,
424            block_param.read_start_line,
425            block_param.data_width,
426            block_param.read_length)
427
428    # Pad igram_block with zeros according to pad_length/pad_width
429    data_block = np.pad(data_block, block_param.block_pad,
430                        mode='constant', constant_values=fill_value)
431
432    return data_block

486    """Replace the value of invalid 'data' cells (indicated by 'invalid')

488    Parameters

511def get_circle_idxs(
512    max_radius: int, min_radius: int = 0, sort_output: bool = True
513) -> np.ndarray:
514    """Get the relative indices of neighboring pixels in a circular pattern.
515
516    This function calculates the relative indices of pixels within a circle
517    defined by a maximum radius. The indices can be sorted for better data
518    access patterns or returned in the order they are found.
519
520    Parameters
521    ----------
522    max_radius : int
523        The maximum radius of the circle.
524    min_radius : int, optional
525        The minimum radius to start calculating indices, default is 0.
526    sort_output : bool, optional
527        Whether to sort the output indices, default is True.
528
529    Returns
530    -------
531    np.ndarray
532        An array of relative pixel indices within the specified circular
533        pattern.
534
535    Notes
536    -----
537    - This function uses a variation of the mid-point circle drawing algorithm
538      to calculate the indices.
539    - If `sort_output` is True, the indices are sorted to improve data access
540      patterns, which can lead to faster runtime.
541    - Adapted from the C++ version of the `psps` package:
542      https://github.com/UT-Radar-Interferometry-Group/psps/blob/a15d458817fe7d06a6edaa0b3208ea78bc4782e7/src/cpp/similarity.cpp#L16
543    """
544    visited = np.zeros((max_radius, max_radius), dtype=bool)
545    visited[0][0] = True
546
547    indices = []
548    for r in range(1, max_radius):
549        x = r
550        y = 0
551        p = 1 - r
552        if r > min_radius:
553            indices.append([r, 0])
554            indices.append([-r, 0])
555            indices.append([0, r])
556            indices.append([0, -r])
557
558        visited[r][0] = True
559        visited[0][r] = True
560        # flag > 0 means there are holes between concentric circles
561        flag = 0
562        while x > y:
563            # do not need to fill holes
564            if flag == 0:
565                y += 1
566                if p <= 0:
567                    # Mid-point is inside or on the perimeter
568                    p += 2 * y + 1
569                else:
570                    # Mid-point is outside the perimeter
571                    x -= 1
572                    p += 2 * y - 2 * x + 1
573            else:
574                flag -= 1
575
576            # All the perimeter points have already been visited
577            if x < y:
578                break
579
580            while not visited[x - 1][y]:
581                x -= 1
582                flag += 1
583
584            visited[x][y] = True
585            visited[y][x] = True
586            if r > min_radius:
587                indices.append([x, y])
588                indices.append([-x, -y])
589                indices.append([x, -y])
590                indices.append([-x, y])
591
592                if x != y:
593                    indices.append([y, x])
594                    indices.append([-y, -x])
595                    indices.append([y, -x])
596                    indices.append([-y, x])
597
598            if flag > 0:
599                x += 1
600
601    if sort_output:
602        # Sorting makes it run faster, better data access patterns
603        return np.array(sorted(indices))
604    else:
605        # Indices run from middle outward
606        return np.array(indices)

609def interpolate_row(row_data):
610    """
611    Interpolate missing or low-weight values in a row of an interferogram.
612
613    This function interpolates values in a row of an interferogram based on the
614    surrounding pixels and their weights. It uses a distance-based weighting
615    scheme to estimate the value of a pixel if its weight is below a specified
616    cutoff. The function is designed to handle complex interferometric data,
617    but currently returns interpolated values as floats.
618
619    Parameters
620    ----------
621    row_data : tuple
622        A tuple containing the following elements:
623        - r0 : int
624            The row index being interpolated.
625        - ifg : numpy.ndarray
626            The 2D array representing the interferogram.
627        - weights : numpy.ndarray
628            The 2D array representing the weights for each pixel.
629        - weight_cutoff : float
630            The cutoff value for weights. Pixels with weights below this value
631            will be interpolated.
632        - num_neighbors : int
633            The maximum number of neighboring pixels to consider for
634            interpolation.
635        - alpha : float
636            The exponent used in the distance weighting function.
637        - indices : list of tuple
638            The list of relative indices for neighboring pixels.
639        - ncol : int
640            The number of columns in the interferogram.
641        - nrow : int
642            The number of rows in the interferogram.
643
644    Returns
645    -------
646    r0 : int
647        The row index that was interpolated.
648    interpolated_row : numpy.ndarray
649        The interpolated row as a 1D array of float32 values.
650    """
651    (r0, ifg, weights, weight_cutoff,
652     num_neighbors, alpha, indices, ncol, nrow) = row_data
653
654    interpolated_row = np.zeros(ncol, dtype=np.float32)
655    for c0 in range(ncol):
656        if weights[r0, c0] < weight_cutoff:
657            csum = 0.0
658            wsum = 0.0
659
660            counter = 0
661            r2 = np.zeros(num_neighbors, dtype=np.float64)
662            phase = np.zeros(num_neighbors, dtype=np.float64)
663
664            for idx in indices:
665                r = r0 + idx[0]
666                c = c0 + idx[1]
667
668                if (0 <= r < nrow) and (0 <= c < ncol) and \
669                   (weights[r, c] >= weight_cutoff):
670                    # Calculate the square distance to the center pixel
671                    r2[counter] = idx[0] ** 2 + idx[1] ** 2
672                    phase[counter] = ifg[r, c]
673                    counter += 1
674                    if counter >= num_neighbors:
675                        break
676
677            if counter > 0:
678                # Normalize and interpolate based on distance and phase
679                r2_norm = (r2[counter - 1] ** alpha) / 2
680                for i in range(counter):
681                    csum += np.exp(-r2[i] / r2_norm) * phase[i]
682                    wsum += np.exp(-r2[i] / r2_norm)
683
684                interpolated_row[c0] = csum / wsum
685            else:
686                interpolated_row[c0] = ifg[r0, c0]
687        else:
688            interpolated_row[c0] = ifg[r0, c0]
689
690    return r0, interpolated_row

761def convolve_preserve_nan(image,
762                          kernel,
763                          mode: str = "constant",
764                          cval: float = 0.0):
765    """
766    Convolve an image with a kernel while preserving NaN values.
767
768    This function performs a convolution on an image, ensuring that NaN
769    values in the original image are preserved in the output. The NaN values
770    are temporarily replaced by zero during the convolution, and the result
771    is normalized based on the coverage of valid (non-NaN) values.
772
773    Parameters
774    ----------
775    image : numpy.ndarray
776        A 2D array representing the input image, where some pixels may be NaN.
777    kernel : numpy.ndarray
778        A 2D array representing the convolution kernel.
779
780    Returns
781    -------
782    convolved_image : numpy.ndarray
783        A 2D array of the same shape as the input image, representing the
784        result of the convolution with NaN values preserved.
785    """
786    # Step 1: Create a mask for valid (non-NaN) values
787    valid_mask = ~np.isnan(image)
788
789    # Step 2: Replace NaN values with zero (or another placeholder)
790    image_filled = np.where(valid_mask, image, 0)
791
792    # Step 3: Apply convolution to the valid mask
793    valid_mask_convolved = convolve(valid_mask.astype(float), kernel,
794                                    mode=mode, cval=cval)
795    # valid_mask_convolved = np.ones_like(image)
796    # Step 4: Apply convolution to the filled image
797    convolved_image = convolve(image_filled,
798                               kernel,
799                               mode=mode,
800                               cval=0.0)
801
802    # Step 5: Normalize the convolution result
803    with np.errstate(invalid='ignore'):
804        convolved_image = convolved_image / valid_mask_convolved
805
806    # Step 6: Restore NaN values in the convolved output
807    convolved_image[~valid_mask] = np.nan
808
809    return convolved_image

852def unwrapping_correction_with_filter(unw,
853                                      kernel_width,
854                                      kernel_length,
855                                      sig_kernel_x,
856                                      sig_kernel_y,
857                                      iterations,
858                                      filter_method):
859    """
860    Correct phase unwrapping errors in an image using iterative filtering.
861
862    This function iteratively applies either a Gaussian convolution or a
863    median filter to correct phase unwrapping errors in an image. It adjusts
864    values modulo 2π during each iteration to correct phase errors.
865
866    Parameters
867    ----------
868    unw : numpy.ndarray
869        A 2D array representing the unwrapped phase image.
870    kernel_width : int
871        The width of the Gaussian kernel.
872    kernel_length : int
873        The length of the Gaussian kernel.
874    sig_kernel_x : float
875        The standard deviation of the Gaussian kernel in the x direction.
876    sig_kernel_y : float
877        The standard deviation of the Gaussian kernel in the y direction.
878    iterations : int
879        The number of iterations for filtering and correction.
880    filter_method : str
881        The method of filtering to use,
882        either 'convolution' or 'median_filter'.
883
884    Returns
885    -------
886    unw : numpy.ndarray
887        The corrected unwrapped phase image.
888    """
889    # Create a mask for zero values in the input
890    mask = unw == 0
891    unw[mask] = np.nan
892
893    # Create and normalize the Gaussian kernel
894    kernel_rows = create_gaussian_kernel(
895        kernel_length, sig_kernel_y).reshape(-1, 1)
896    kernel_cols = create_gaussian_kernel(
897        kernel_width, sig_kernel_x).reshape(1, -1)
898    kernel_norm = (kernel_rows * kernel_cols) / np.sum(
899        kernel_rows * kernel_cols)
900
901    # Iteratively apply the selected filter and correct phase unwrapping errors
902    for _ in range(iterations):
903        if filter_method == 'convolution':
904            filtered_img = convolve_preserve_nan(unw, kernel_norm)
905        elif filter_method == 'median_filter':
906            filtered_img = nan_median_filter(unw, size=kernel_width)
907        else:
908            raise ValueError("Invalid filter_method. Choose 'convolution' or 'median_filter'.")
909
910        # Restore NaN values in the filtered image and correct phase errors
911        filtered_img[mask] = np.nan
912        diff = unw - filtered_img
913        unw_err_ind = np.round(diff / (2 * np.pi))
914        unw -= unw_err_ind * 2 * np.pi
915
916    # Restore original zero values in the output image
917    unw[mask] = 0
918    return unw

518    defined by a maximum radius. The indices can be sorted for better data

539      to calculate the indices.

541      patterns, which can lead to faster runtime.

546    visited[0][0] = True
547

550        x = r
551        y = 0
552        p = 1 - r
553        if r > min_radius:
554            indices.append([r, 0])
555            indices.append([-r, 0])

557            indices.append([0, -r])
558

568                    # Mid-point is inside or on the perimeter
569                    p += 2 * y + 1
570                else:

572                    x -= 1
573                    p += 2 * y - 2 * x + 1
574            else:
575                flag -= 1
576
577            # All the perimeter points have already been visited
578            if x < y:
579                break
580
581            while not visited[x - 1][y]:

583                flag += 1
584
585            visited[x][y] = True

589                indices.append([-x, -y])
590                indices.append([x, -y])

595                    indices.append([-y, -x])
596                    indices.append([y, -x])
