  9def preprocess_wrapped_igram(igram, coherence, mask=None,
 10                             mask_type='coherence', threshold=0.5,
 11                             filter_size=9,
 12                             filling_enabled=True,
 13                             filling_method='distance_interpolator',
 14                             distance=5):
 15    '''
 16    Preprocess wrapped interferograms prior to phase unwrapping.
 17
 18    Removes invalid pixels in wrapped interferograms based on
 19    user-defined metric. Invalid pixels may be identified using
 20    1) a water mask; 2) thresholding low-coherence pixels; 3) thresholding
 21    the median absolute deviation of the interferogram phase from the local median.
 22    Invalid pixels are replaced with values computed with a distance-weighted
 23    interpolation approach from Chen et al., 2015. The magnitude of the complex
 24    interferogram is discarded.
 25
 26    Parameters
 27    ----------
 28    igram: numpy.ndarray
 29        Wrapped interferogram to pre-process
 30    coherence: numpy.ndarray
 31        Normalized InSAR coherence
 32    mask: numpy.ndarray or None
 33        Optional binary mask (1: invalid; 0: valid) to identify invalid pixels.
 34        If a mask is provided, data-driven masking is not performed (other
 35        masking options are ignored;
 36    mask_type: str, {'median_filter', 'coherence', 'water'}, optional
 37        Type of mask to identify invalid pixels
 38        'median_filter':
 39        Compute mask of invalid pixels by thresholding the median absolute
 40        deviation w.r.t. the local neighborhood around each pixel.
 41
 42        'coherence':
 43        The default mode. Compute mask of invalid pixels by thresholding
 44        the normalized InSAR coherence.
 45
 46        'water':
 47        Project the water mask to radar grid and masks out the invalid
 48        pixels
 49
 50    threshold: float
 51        Threshold to identify invalid pixels.
 52        If 'mask_type' is 'coherence' pixels with coherence below threshold
 53        are considered invalid
 54        If 'mask_type' is 'median_filter' pixels with median absolute
 55        deviation (MAD) above this threshold are considered outliers
 56    filter_size: int
 57        Size of median filter for median absolute deviation
 58        outlier identification method
 59    filling_method: str
 60        Algorithm to fill invalid pixels. 'distance_interpolator'
 61        applies distance weighted interpolation from Chen et al., 2015
 62    distance: int
 63        Distance metric for interpolation. For distance interpolator in
 64        Chen et al [1]_ is distance is intended as radius
 65
 66    Returns
 67    -------
 68    filt_igram: numpy.ndarray
 69        Wrapped interferogram with outlier pixel being filtered
 70        out and replaced with pixels computed by the selected
 71        'filling_method'. The magnitude of the input wrapped interferogram
 72        is discarded.
 73
 74    References
 75    ----------
 76    .. [1] J. Chen, H. A. Zebker,and R. Knight, ""A persistent scatterer interpolation
 77       for retrieving accurate ground deformation over InSAR-decorrelated
 78       Agricultural fields", Geoph. Res. Lett., 42(21), 9294-9301, (2015).
 79    '''
 80
 81    # Extract some preprocess options
 82    error_channel = journal.error('unwrap.run.preprocess_wrapped_igram')
 83
 84    # Create mask of invalid pixels
 85    invalid_mask = np.full(igram.shape, dtype=bool, fill_value=False)
 86
 87    # Identify invalid pixels and store them in a mask.
 88    # Criteria to identify invalid pixels:
 89    # 1-1) Based on user-provided mask
 90    # 1-2) Based on water mask
 91    if mask is not None:
 92        invalid_mask[mask == 1] = True
 93    # 2) Based on InSAR correlation values
 94    elif mask_type == 'coherence':
 95        invalid_mask[coherence < threshold] = True
 96    # 3) Based on median absolute deviation (MAD)
 97    elif mask_type == 'median_filter':
 98        igram_pha = np.angle(igram)
 99        mad = median_absolute_deviation(igram_pha, filter_size)
100        invalid_mask[mad > threshold] = True
101    # Not a valid algorithm to mask pixels
102    else:
103        err_str = f"{mask_type} is an invalid selection for mask_type"
104        error_channel.log(err_str)
105        raise ValueError(err_str)
106
107    if filling_enabled:
108        # Fill invalid interferogram pixels using user-defined algorithm
109        # Distance-based interpolator Chen et al. _[1]
110        if filling_method == 'distance_interpolator':
111            phase_filt = distance_interpolator(np.angle(igram), distance,
112                                            invalid_mask)
113        else:
114            err_str = f"{filling_method} is an invalid selection for filling_method"
115            error_channel.log(err_str)
116            raise ValueError(err_str)
117    else:
118        igram[invalid_mask==1] = 0
119        phase_filt = np.angle(igram)
120    # Go to complex value
121    igram_filt = np.exp(1j * phase_filt)
122
123    return igram_filt

126def distance_interpolator(arr, radius, invalid_mask):
127    '''
128    Interpolate pixels based on distance from valid pixels
129    following Chen et al [1]_.
130
131    Parameters
132    ----------
133    arr: numpy.ndarray
134        Array containing invalid pixel locations to fill
135    radius: int
136        Radius of the sampling/filling window
137    invalid_mask: numpy.ndarray
138        Boolean mask identifying invalid pixels (True:invalid)
139
140    Returns
141    -------
142    fill_arr: numpy.ndarray
143        Array with interpolated values at invalid pixel locations
144
145    References
146    __________
147    .. [1] J. Chen, H. A. Zebker,and R. Knight, ""A persistent scatterer interpolation
148       for retrieving accurate ground deformation over InSAR-decorrelated
149       Agricultural fields", Geoph. Res. Lett., 42(21), 9294-9301, (2015).
150    '''
151    arr_filt = np.copy(arr)
152
153    # Get center locations
154    x_cent, y_cent = np.where(invalid_mask == True)
155
156    # Find the coordinates of valid pixels
157    x, y = np.where(invalid_mask == False)
158
159    for xc, yc in zip(x_cent, y_cent):
160        # Compute distance between center pixel and valid pixels
161        ps_dist = np.sqrt((x - xc) ** 2 + (y - yc) ** 2)
162        # Compute weights based on distance and selected radius
163        w = np.exp(-ps_dist ** 2 / 2 * radius)
164        # Compute Eq. 2 of Chen at al [1]_
165        weighted_arr = arr_filt[x, y].flatten() * w
166        arr_filt[xc, yc] = np.nansum(weighted_arr) / np.nansum(w)
167
168    return arr_filt

171def median_absolute_deviation(arr, filter_size):
172    '''
173    Compute the median absolute deviation (MAD) of `arr`
174    defined as median(abs(arr - median(arr))
175
176    Parameters
177    ----------
178    arr: numpy.ndarray
179        Array for which to compute MAD
180    filter_size: int
181        Size of median filter, in pixels
182
183    Returns
184    -------
185    mad: numpy.ndarray
186        Median absolute deviation of `arr`
187    '''
188    med = np.abs(arr - median_filter(arr, [filter_size, filter_size]))
189    mad = median_filter(med, [filter_size, filter_size])
190    return mad

225def _read_gdal_with_bbox(gdal_raster, bbox):
226    '''
227    Extract image from the gdal-supported file with bbox
228
229    Parameters
230    ----------
231    gdal_raster: osgeo.gdal.Dataset
232        gdal dataset to extract the subset image
233    bbox: list
234        list of [xmin, ymin, xmax, ymax]
235
236    Returns
237    -------
238    subset_data: numpy.ndarray
239        Median absolute deviation of `arr`
240    [sub_x0, sub_y0, dx, dy]: list
241        sub_x0: x coordinate of upper left
242        sub_y0: y coordinate of upper left
243        dx: x spacing
244        dy: y spacing
245    '''
246    xmin, ymin, xmax, ymax = bbox
247
248    geotransform = gdal_raster.GetGeoTransform()
249    x0 = geotransform[0]
250    y0 = geotransform[3]
251    dx = geotransform[1]
252    dy = geotransform[5]
253
254    idx_start = int(np.floor((xmin - x0) / dx))
255    idx_end = int(np.ceil((xmax - x0) / dx))
256
257    if dy > 0:
258        idy_start = int(np.floor((ymin - y0) / dy))
259        idy_end = int(np.ceil((ymax - y0) / dy))
260        sub_y0 = idy_start*dy + y0
261    else:
262        idy_start = int(np.floor((ymax - y0) / dy))
263        idy_end = int(np.ceil((ymin - y0) / dy))
264
265    idx_start = max(0, idx_start)
266    idy_start = max(0, idy_start)
267
268    x_width = idx_end - idx_start
269    y_length = idy_end - idy_start
270
271    if x_width > gdal_raster.RasterXSize:
272        x_width = gdal_raster.RasterXSize
273    if y_length > gdal_raster.RasterYSize:
274        y_length = gdal_raster.RasterYSize
275
276    sub_x0 = idx_start*dx + x0
277    sub_y0 = idy_start*dy + y0
278    raster_band = gdal_raster.GetRasterBand(1)
279    subset_data = raster_band.ReadAsArray(idx_start,
280                                          idy_start,
281                                          x_width,
282                                          y_length)
283
284    return subset_data, [sub_x0, sub_y0, dx, dy]

287def project_map_to_radar(cfg, input_data_path, freq):
288    '''
289    Project map coordinate image to radar grid
290
291    Parameters
292    ----------
293    cfg: dict
294        input runconfig file
295    input_data_path: str
296        input file path for map coordinate image
297    freq: str
298        frequency to be projected
299
300    Returns
301    -------
302    rdr_data: numpy.ndarray
303        projected data into radar grid  absolute
304    '''
305    scratch_path = pathlib.Path(cfg['product_path_group']['scratch_path'])
306    rdr2geo_path = f'{scratch_path}/rdr2geo'
307
308    az_looks = cfg["processing"]["crossmul"]["azimuth_looks"]
309    rg_looks = cfg["processing"]["crossmul"]["range_looks"]
310    unw_az_looks = cfg["processing"]["phase_unwrap"]["azimuth_looks"]
311    unw_rg_looks = cfg["processing"]["phase_unwrap"]["range_looks"]
312    if unw_az_looks != 1:
313        az_looks = unw_az_looks
314    if unw_rg_looks != 1:
315        rg_looks = unw_rg_looks
316
317    # prepare input paths
318    topo_paths = {xy: f'{rdr2geo_path}/freq{freq}/{xy}.rdr' for xy in 'xy'}
319
320    # get input shape and type - input type also output type
321    _, output_dtype = _get_gdal_raster_shape_type(input_data_path)
322    geo_data_raster = gdal.Open(input_data_path)
323
324    # for both x and y rasters, decimate and get extents
325    decimated_blocks = {}
326    decimated_extents = {}
327    for xy, input_path in topo_paths.items():
328        # open input raster for reading
329        input_data_raster = gdal.Open(input_path)
330        input_data = input_data_raster.ReadAsArray()
331        rows, cols = input_data.shape
332        # if multi-looks are 1 or 2,
333        # slice_az_end and slice_rg_end are 0. To avoid the positive
334        # number, we take None.
335        az_size = rows // az_looks
336        rg_size = cols // rg_looks
337        slice_az_start = int(az_looks / 2)
338        slice_az_end = az_size * az_looks
339        slice_rg_start = int(rg_looks / 2)
340        slice_rg_end = rg_size * rg_looks
341
342        # take center pixels of block to decimate
343        decimated_arr = \
344            input_data[slice_az_start:slice_az_end:az_looks,
345                       slice_rg_start:slice_rg_end:rg_looks]
346
347        # save decimated extents and array for current axis
348        decimated_extents[xy] = [np.nanmin(decimated_arr),
349                                 np.nanmax(decimated_arr)]
350        decimated_blocks[xy] = decimated_arr
351        del input_data
352
353    # get bounding for decimated extents
354    bbox = [decimated_extents['x'][0], decimated_extents['y'][0],
355            decimated_extents['x'][1], decimated_extents['y'][1]]
356
357    # read map bounded by decimated extents of xy block
358    input_arr_block, [block_x0, block_y0, block_dx, block_dy] = \
359        _read_gdal_with_bbox(geo_data_raster, bbox)
360
361    # prepare output array
362    output_arrays = np.zeros(decimated_blocks['y'].shape,
363                             dtype=output_dtype)
364
365    # prepare coordinates to map to
366    coordinates = ((decimated_blocks['y'] - block_y0) / block_dy,
367                   (decimated_blocks['x'] - block_x0) / block_dx)
368    # map input raster to decimated coordinates
369    map_coordinates(input_arr_block,
370                    coordinates,
371                    output=output_arrays,
372                    mode='nearest',
373                    order=0,
374                    cval=np.nan,
375                    prefilter=False)
376
377    # stack to make whole then return
378    return output_arrays

315    if unw_rg_looks != 1:
316        rg_looks = unw_rg_looks
317
318    # prepare input paths
319    topo_paths = {xy: f'{rdr2geo_path}/freq{freq}/{xy}.rdr' for xy in 'xy'}
320
321    # get input shape and type - input type also output type
322    _, output_dtype = _get_gdal_raster_shape_type(input_data_path)
323    geo_data_raster = gdal.Open(input_data_path)
324
325    # for both x and y rasters, decimate and get extents
326    decimated_blocks = {}
327    decimated_extents = {}
328    for xy, input_path in topo_paths.items():
329        # open input raster for reading
330        input_data_raster = gdal.Open(input_path)
331        input_data = input_data_raster.ReadAsArray()
332        rows, cols = input_data.shape
333        # if multi-looks are 1 or 2,
334        # slice_az_end and slice_rg_end are 0. To avoid the positive
335        # number, we take None.
336        az_size = rows // az_looks
337        rg_size = cols // rg_looks
338        slice_az_start = int(az_looks / 2)
339        slice_az_end = az_size * az_looks
340        slice_rg_start = int(rg_looks / 2)
341        slice_rg_end = rg_size * rg_looks
342
343        # take center pixels of block to decimate
344        decimated_arr = \
345            input_data[slice_az_start:slice_az_end:az_looks,
346                       slice_rg_start:slice_rg_end:rg_looks]
347
348        # save decimated extents and array for current axis
349        decimated_extents[xy] = [np.nanmin(decimated_arr),
350                                 np.nanmax(decimated_arr)]
351        decimated_blocks[xy] = decimated_arr
352        del input_data
353
354    # get bounding for decimated extents
355    bbox = [decimated_extents['x'][0], decimated_extents['y'][0],
356            decimated_extents['x'][1], decimated_extents['y'][1]]
357
358    # read map bounded by decimated extents of xy block
359    input_arr_block, [block_x0, block_y0, block_dx, block_dy] = \
360        _read_gdal_with_bbox(geo_data_raster, bbox)
361
362    # prepare output array
363    output_arrays = np.zeros(decimated_blocks['y'].shape,
364                             dtype=output_dtype)
365
366    # prepare coordinates to map to
367    coordinates = ((decimated_blocks['y'] - block_y0) / block_dy,
368                   (decimated_blocks['x'] - block_x0) / block_dx)
369    # map input raster to decimated coordinates
370    map_coordinates(input_arr_block,
371                    coordinates,
372                    output=output_arrays,
373                    mode='nearest',
374                    order=0,
375                    cval=np.nan,
376                    prefilter=False)
377
378    # stack to make whole then return
379    return output_arrays
